using System.ServiceProcess;
using System.Timers;
using DigitalSignatureApp;

namespace DigitalSignatureApplication
{
    public partial class dscsync : ServiceBase
    {
        Timer timer = new Timer();
        public bool IsTriggerRunning = false;
        public dscsync()
        {
            this.ServiceName = "Digital Signature";
            InitializeComponent();
        }

        protected override void OnStart(string[] args)
        {
            this.timer = new System.Timers.Timer(10000D);  // 30000 milliseconds = 30 seconds
            this.timer.AutoReset = true;
            this.timer.Elapsed += new System.Timers.ElapsedEventHandler(this.timer_Elapsed);
            this.timer.Start();
        }

        protected override void OnStop()
        {
            this.timer.Stop();
            this.timer = null;
        }
        private async void timer_Elapsed(object sender, System.Timers.ElapsedEventArgs e)
        {
            Application processInitiation = new Application();
            await processInitiation.Sequence();
        }
    }
}
